# CarShop Pazoo 入れ替え用 301リダイレクト — 2026-09-09 太郎
# 置き場所: pazoo.co.jp のドキュメントルート直下 .htaccess
# 旧URL一覧は 2026-09-09 に pazoo.co.jp/sitemap.xml を実測して作成（旧11URL）。
# ★現行 sitemap の lastmod は 2022-12-01（約6年放置の裏付け）。

<IfModule mod_rewrite.c>
RewriteEngine On

# --- 階層が深いものを先に書く（順序が逆だと /order/ に飲まれる）---
RewriteRule ^order/insurance/?$   /order-insurance.html  [R=301,L]

# --- 旧WordPress（ディレクトリ型） → 新（.html）---
RewriteRule ^coating/?$           /coating.html          [R=301,L]
RewriteRule ^companyoutline/?$    /companyoutline.html   [R=301,L]
RewriteRule ^contact/?$           /contact.html          [R=301,L]
RewriteRule ^maintenance/?$       /maintenance.html      [R=301,L]
RewriteRule ^order/?$             /order.html            [R=301,L]
RewriteRule ^privacy/?$           /privacy.html          [R=301,L]
RewriteRule ^repair-paint/?$      /repair-paint.html     [R=301,L]

# --- テーマ由来の重複URL ---
RewriteRule ^lightning-ja/contact/?$  /contact.html      [R=301,L]

# --- 🔴 社長承認待ち: /used-car/ の行き先 -------------------------
# 新サイトに中古車在庫ページが無い（在庫は goo-net へリンクする方針）。
# 太郎の推し = /order.html（注文販売・買取＝いちばん意味が近い）。
# 別案 = / （トップ）。承認後にどちらかを残して1行にする。
RewriteRule ^used-car/?$          /order.html            [R=301,L]
# ---------------------------------------------------------------

RewriteRule ^sitemap-page/?$      /                      [R=301,L]

# --- WordPressの残骸 ---
RewriteRule ^feed/?$              /                      [R=301,L]
RewriteRule ^wp-content/          /                      [R=301,L]
RewriteRule ^wp-includes/         /                      [R=301,L]
</IfModule>
